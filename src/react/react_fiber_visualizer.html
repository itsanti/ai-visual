Исходный компонент

JSX · пример

import { useEffect, useState, useTransition } from 'react';

let nextId = 3;
function ItemList() {
  const [items, setItems] = useState([
    { id: 1, label: 'Элемент 1' },
    { id: 2, label: 'Элемент 2' }
  ]);
  const [isPending, startTransition] = useTransition();

  useEffect(() => {
    console.log('setup', items.length);
    return () => console.log(
      'cleanup', items.length
    );
  }, [items.length]);

  function addItem() {
    const id = nextId++;
    setItems(prev => [
      ...prev, { id, label: `Элемент ${id}` }
    ]);
  }

  function addInTransition() {
    startTransition(() => addItem());
  }

  return (
    <div>
      <button onClick={addItem}>Добавить</button>
      <button onClick={addInTransition}>Через useTransition</button>
      <p role="status">{isPending ? 'Обновляем список…' : ''}</p>
      <ul>
        {items.map(item => (
          <li key={item.id}>{item.label}</li>
        ))}
      </ul>
    </div>
  );
}
Подсвечиваются строки, относящиеся к текущему этапу.

Почему useTransition не убирает фризы от тяжёлого CPU-кода

Колбэк выполняется сразу и синхронно. Вызов startTransition(() => setItems(expensiveCalc())) сначала вычислит expensiveCalc() в основном потоке. Пока функция работает, браузер не может обработать ввод или отрисовать кадр. Даже индикатор isPending появится только после освобождения потока.

React может уступать основной поток между единицами работы Fiber во время render, например при обработке множества отдельных компонентов. Один долгий вызов компонента или большой синхронный map() внутри него выполняется целиком. Когда React доходит до commit, он синхронно применяет изменения DOM. Большой commit тоже может задержать кадр.

Границы модели. Симулятор показывает один возможный порядок работы React. Два commit могут попасть в один кадр браузера. Здесь опущены детали lanes, batching, bailout и перезапуска render. useEffect выполняется после commit, а его точный момент относительно отрисовки может меняться.
